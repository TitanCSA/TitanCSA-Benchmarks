cmake_minimum_required(VERSION 3.24)
project(TitanCSABenchmarks LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
 message(FATAL_ERROR "Windows and MSVC required for native measurements")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
add_compile_options(/W4 /EHsc /utf-8 /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
set(VDEVICE_ROOT "${CMAKE_SOURCE_DIR}/.local/sources/TitanCSA--Vdevice" CACHE PATH "Pinned Vdevice source checkout")
add_library(VdeviceSDK STATIC "${VDEVICE_ROOT}/Vdevice/Vdevice.cpp")
target_include_directories(VdeviceSDK PUBLIC "${VDEVICE_ROOT}/Vdevice")
add_executable(Vdevice-Upstream-LiveTests "${VDEVICE_ROOT}/Tests/ExampleLiveTests.cpp")
target_link_libraries(Vdevice-Upstream-LiveTests PRIVATE VdeviceSDK setupapi hid xinput advapi32 user32)
add_executable(TitanCSA-Benchmark-InputGenerator benchmarks/input-generator/main.cpp)
target_link_libraries(TitanCSA-Benchmark-InputGenerator PRIVATE VdeviceSDK setupapi hid psapi)
add_executable(TitanCSA-Benchmark-HidReceiver benchmarks/input-receiver/hid.cpp)
target_link_libraries(TitanCSA-Benchmark-HidReceiver PRIVATE hid psapi)
add_executable(TitanCSA-Benchmark-RawReceiver benchmarks/input-receiver/rawinput.cpp)
target_link_libraries(TitanCSA-Benchmark-RawReceiver PRIVATE user32 psapi)
add_executable(TitanCSA-Benchmark-XInputReceiver benchmarks/input-receiver/xinput.cpp)
target_link_libraries(TitanCSA-Benchmark-XInputReceiver PRIVATE xinput psapi)
add_executable(TitanCSA-Benchmark-Ds4Receiver benchmarks/input-receiver/ds4.cpp)
target_link_libraries(TitanCSA-Benchmark-Ds4Receiver PRIVATE setupapi hid psapi)
set(VIGEM_ROOT "${CMAKE_SOURCE_DIR}/.local/sources/nefarius--ViGEmClient" CACHE PATH "Pinned ViGEmClient source")
add_subdirectory("${VIGEM_ROOT}" "${CMAKE_BINARY_DIR}/ViGEmClient" EXCLUDE_FROM_ALL)
add_executable(TitanCSA-Benchmark-XInput benchmarks/virtual-devices/xinput.cpp)
# Stable descriptor serial allows Windows to persist its per-instance filter
# association across the required guest restart. Only the fixture identity is
# adapted; the upstream SDK, driver and report handling remain unchanged.
file(READ "${VDEVICE_ROOT}/Examples/UI/ExampleRuntime.h" FIXTURE_RUNTIME)
set(SERIAL_OLD "const auto serial=\"vd-\"+std::to_string(GetCurrentProcessId())+\"-\"+std::to_string(counter.QuadPart);")
set(SERIAL_NEW "const auto serial=\"TitanCSABenchmarkX360StableV1\";")
string(FIND "${FIXTURE_RUNTIME}" "${SERIAL_OLD}" SERIAL_POSITION)
if(SERIAL_POSITION EQUAL -1)
 message(FATAL_ERROR "Pinned example serial anchor changed; inspect fixture adaptation")
endif()
string(REPLACE "${SERIAL_OLD}" "${SERIAL_NEW}" FIXTURE_RUNTIME "${FIXTURE_RUNTIME}")
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/ExampleRuntime.h" "${FIXTURE_RUNTIME}")
target_include_directories(TitanCSA-Benchmark-XInput PRIVATE "${CMAKE_BINARY_DIR}/generated" "${VDEVICE_ROOT}/Examples/UI")
target_link_libraries(TitanCSA-Benchmark-XInput PRIVATE VdeviceSDK ViGEmClient setupapi hid psapi advapi32)
foreach(edition IN ITEMS Keyboard-Mouse Combined Gamepad-Controller)
 if(edition STREQUAL "Combined")
  set(SDK_ROOT "${CMAKE_SOURCE_DIR}/.local/sources/TitanCSA--Interceptor")
 elseif(edition STREQUAL "Keyboard-Mouse")
  set(SDK_ROOT "${CMAKE_SOURCE_DIR}/.local/sources/TitanCSA--Interceptor-Keyboard-Mouse")
 else()
  set(SDK_ROOT "${CMAKE_SOURCE_DIR}/.local/sources/TitanCSA--Interceptor-Gamepad-Controller")
 endif()
 add_executable("TitanCSA-Benchmark-Intercept-${edition}" benchmarks/common/intercept.cpp "${SDK_ROOT}/Interceptor/Interceptor.cpp")
 target_include_directories("TitanCSA-Benchmark-Intercept-${edition}" PRIVATE "${SDK_ROOT}/Interceptor")
 target_compile_options("TitanCSA-Benchmark-Intercept-${edition}" PRIVATE /FIwindows.h /FIwinioctl.h)
 target_link_libraries("TitanCSA-Benchmark-Intercept-${edition}" PRIVATE setupapi hid newdev cfgmgr32 advapi32 user32 psapi)
endforeach()
set(VJOY_SDK "${CMAKE_SOURCE_DIR}/.local/vjoy-sdk/SDK" CACHE PATH "Pinned vJoy SDK")
if(EXISTS "${VJOY_SDK}/inc/vjoyinterface.h")
 add_executable(TitanCSA-Benchmark-Joystick benchmarks/virtual-devices/joystick.cpp)
 target_include_directories(TitanCSA-Benchmark-Joystick PRIVATE "${VJOY_SDK}/inc")
 target_link_libraries(TitanCSA-Benchmark-Joystick PRIVATE VdeviceSDK psapi)
 add_executable(TitanCSA-Benchmark-JoystickReceiver benchmarks/input-receiver/joystick.cpp)
 target_link_libraries(TitanCSA-Benchmark-JoystickReceiver PRIVATE winmm psapi)
endif()
